\begin{frame}{Method: sample, synthesize, score, then route the budget}
  \centering
  \begin{tikzpicture}[
      >=Stealth,
      node distance=0.72cm and 0.58cm,
      box/.style={draw=positive!70!black,fill=white,rounded corners=3pt,
                  minimum height=1.05cm,text width=2.15cm,align=center,inner sep=4pt},
      agg/.style={box,draw=negative!85!black,fill=softorange,text width=3.1cm},
      score/.style={box,draw=emphasis!80!black,fill=white,text width=2.15cm},
      route/.style={draw=cbPurple!85!black,fill=cbPurple!12,rounded corners=3pt,
                    text width=2.25cm,align=center,inner sep=4pt}
    ]
    \node[box] (input) {Prompt $x$};
    \node[box,right=of input] (sample) {Sample\\$y^{(1)},\ldots,y^{(K)}$};
    \node[agg,right=of sample] (aggregate) {LLM synthesis\\$\hat y=\mathcal{A}(\{y^{(k)}\}_{k=1}^{K})$};
    \node[score,right=of aggregate] (fact) {FactOWL\\atomic-fact score};
    \node[route,below=0.72cm of sample] (adaptive) {Pre-generation\\router $K=g(x)$};
    \draw[->,thick] (input) -- (sample);
    \draw[->,thick] (sample) -- (aggregate);
    \draw[->,thick] (aggregate) -- (fact);
    \draw[->,thick,draw=cbPurple!85!black] (adaptive) -- node[left,font=\footnotesize] {budget} (sample);
    \node[font=\small,align=center,text width=5.2cm,below=0.35cm of fact] {Goal: preserve factuality; avoid unnecessary generations};
  \end{tikzpicture}
  \vfill
  \begin{itemize}
    \item \textbf{Single:} one completion, no aggregation.
    \item \textbf{LLMAgg:} merge consistent claims and remove contradictions.
    \item \textbf{Adaptive:} choose $K$ before any candidate is generated.
  \end{itemize}
  \note{Key message: the intervention is content-level synthesis plus a separate pre-generation budget policy. Transition: why synthesis matters becomes clearer when candidate answers disagree.}
\end{frame}

  \vspace{7pt}
  \begin{alertblock}{Aggregation objective}
    Preserve supported, complementary facts; omit unsupported claims and contradictions.
  \end{alertblock}
  \begin{itemize}
    \item Selection can only inherit the errors of one existing response.
    \item \LLMAgg{} writes a new response from the pooled evidence.
  \end{itemize}
  \note{Key message: this is a precision story, not a recall or oracle-bound story. The best sampled response is a diagnostic; synthesis can produce a stronger new answer. Transition: compare that mechanism with every baseline explicitly.}

\begin{frame}{Adaptive routing uses pre-generation complexity signals}
  \begin{columns}[T,onlytextwidth]
    \column{0.43\textwidth}
      \begin{block}{Controller}\small
        Random forest predicts a budget $K=g(x)$ before generation.
        \begin{itemize}
          \item 300 trees; minimum leaf size 3.
          \item Prompt length, lexical complexity, punctuation, years.
          \item Language/domain indicators and a domain budget prior.
          \item No partial output or disagreement features.
        \end{itemize}
      \end{block}
      \vspace{5pt}
      \centering
      \textbf{Length + average word length} account for over 60\% of feature importance.
    \column{0.54\textwidth}
      \includegraphics[width=0.96\textwidth]{figures/adaptive_feature_importance.pdf}
  \end{columns}
  \note{Key message: the router is deliberately simple and pre-generation. It reads input complexity, not model confidence or candidate disagreement. Transition: evaluate the practical tradeoff against the conservative 99-sample reference.}
\end{frame}

\begin{frame}{Adaptive routing saves against $K=99$, not $K=20$}
  \begin{columns}[T,onlytextwidth]
    \column{0.47\textwidth}
      \begin{block}{$K=20$ is a different reference}
        \begin{itemize}
          \item Budget curves identify $K=20$ as near-saturated.
          \item Against $K=20$, the controller produces no net savings.
          \item Predicted budgets average 70.6--96.9 samples.
        \end{itemize}
      \end{block}
      \vspace{5pt}
      \centering
      \textbf{Savings depend on the reference budget.}
    \column{0.47\textwidth}
      \begin{alertblock}{Generalization and safety}
        \begin{itemize}
          \item Held-out-domain coverage drops to roughly 0.24--0.65.
          \item Routing does not detect inputs where LLMAgg degrades.
          \item One generator family; evaluator noise remains possible.
        \end{itemize}
      \end{alertblock}
      \vspace{5pt}
      \centering
      \textbf{Same-distribution routing, not a domain-agnostic recipe.}
  \end{columns}
  \note{Key message: the strongest caveat is reference sensitivity. The method compresses K=99, but does not beat a hand-calibrated K=20 baseline or solve catastrophic aggregation failures. Transition: close by separating what the evidence supports from what remains open.}
\end{frame}

\begin{frame}{Three takeaways}
  \begin{enumerate}
    \item \textbf{Synthesis beats selection for long-form factual precision.}\\
          \small The main gain comes from consolidating complementary facts, not choosing a single ``best'' completion.
    \vspace{6pt}
    \item \textbf{Sampling gains saturate early, but saturation is heterogeneous.}\\
          \small Domain and language determine how much additional computation remains useful.
    \vspace{6pt}
    \item \textbf{Adaptive routing reduces the $K=99$ budget; it does not beat $K=20$.}\\
          \small It saves 2--29\% against 99 samples, while $K=20$ sensitivity and failure detection remain open.
  \end{enumerate}
  \vfill
  \begin{block}{Answer to the opening tension}
    \centering
    \large Better consolidation matters more than simply generating more answers.
  \end{block}
  \note{Key message: the paper supports a practical hierarchy: aggregate well first, then calibrate and route the budget. Transition: references, then questions.}
\end{frame}

\begin{frame}{Adaptive policy: definitions and training protocol}
  \begin{columns}[T,onlytextwidth]
    \column{0.52\textwidth}
      \begin{block}{Reported quantities}\small
        \begin{align*}
          \text{compute savings}
            &= 1-\frac{\sum_i K_i^{\mathrm{adaptive}}}{N\,\Kfull}\\[-2pt]
          \text{score retention}
            &= \frac{\mathrm{Fact}(\hat y_{\mathrm{adaptive}})}
                    {\mathrm{Fact}(\hat y_{\mathrm{full}})}\\[-2pt]
          \text{coverage}
            &= \Pr[\text{retention meets target}]
        \end{align*}
      \end{block}

      \begin{block}{Protocol}\small

          \item Five stratified shuffle splits.
          \item Test fraction: 30\%.
          \item Split seed: 13.
          \item Model seed: 13 + fold index.
          \item Targets: 0.95 and 0.99.
        \end{itemize}
      \end{block}
  \end{columns}
